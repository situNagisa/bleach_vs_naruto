```
// ============================================================
// 原语：N 个静态已知类型的构造顺序问题（路线 B：递归发现依赖）
// ============================================================

// -------- 编译期：类型在参数包里的下标，不需要运行时查找 --------
pack_index<Target, T...>: size_t
    = 0                                          if Target == T[0]
    = 1 + pack_index<Target, T[1..]>             otherwise
    // Target 不在 T... 里 => 编译错误（找不到这个特化/重载）

// -------- catalog：N 个类型的构造顺序原语 --------
// 继承 N 份 manual_lifetime<T>，一个类型一个专属的存储槎位。
// state 不挂在每个槎位里，集中存成两个 bitset，避免每槎位因为
// 对齐要求产生 padding；两个 bitset 都只有 N 位，通常整体常驻缓存。
catalog<T...> : manual_lifetime<T>...
    built:     bitset<sizeof...(T)> = 全 0
    building:  bitset<sizeof...(T)> = 全 0

    // 可选：构造 catalog 本身时，把某几个"根"节点提前种好、标记 built，
    // 其余节点保持 unbuilt。这是唯一允许"额外参数"的地方——因为 catalog
    // 只构造一次，不存在"先后调用给不同参数"的记忆化冲突。
    // 具体怎么种、种哪几个，由使用者自己在自己的构造函数里决定，不是
    // catalog 这个原语规定的固定接口。

    get<Target>() -> Target&:
        index ← pack_index<Target, T...>        // 编译期常量

        if built[index]:
            return this.manual_lifetime<Target>.get()

        assert not building[index]              // 成环，当场断言
        building[index] ← true

        try:
            // 直接用 Target 自己的构造函数原地构造，*this 是
            // catalog&，Target 的构造函数体里可以递归调 self.get<U>()。
            this.manual_lifetime<Target>.construct_from(
                () => Target(self = *this))
        catch:
            building[index] ← false             // 允许下次重试
            rethrow

        building[index] ← false
        built[index] ← true
        return this.manual_lifetime<Target>.get()

    ~catalog():
        // 按真实构造顺序的逆序销毁（不是 T... 的声明顺序）；
        // 需要另记一条构造顺序的列表，销毁时逆着走。
        for target in 真实构造顺序.reverse():
            this.manual_lifetime<target>.destroy()


// ============================================================
// 用法：entity 跨帧 / task 单帧，两层递归嵌套同一个原语
// ============================================================

// -------- 长期存活的对象：调用者持有，永远不进任何 catalog --------
camera:   { ... 跨帧不变的状态 ... }
foliage:  { ... 跨帧不变的状态 ... }
renderer: { ... 跨帧不变的状态 ... }

// -------- 每一帧：外层 catalog，节点 = "某个 entity 这一帧的任务集合" --------

camera_frame:
    entity: camera*                              // 外层 catalog 构造时种进来
    tasks:  catalog<view_task>                    // 内层 catalog：这个 entity 这一帧的 task

    camera_frame(self: outer_catalog&):
        entity ← /* 从 self 构造时种下的根信息里取 */
        tasks  ← catalog<view_task>{}             // 空的，view_task 还没造，等被 get 到才造

view_task:
    sender: some_sender_type

    view_task(self: catalog<view_task>&):
        cam_frame ← /* 这一层没有别的节点可依赖，直接用 entity 的状态 */
        sender ← compute_view(cam_frame.entity 的状态)


foliage_frame:
    record: optional<record_task>                 // 缺席用值本身表达，不用 find

    foliage_frame(self: outer_catalog&):
        cam_frame ← self.get<camera_frame>()       // 递归拉起 camera 这一帧的任务集合
        view      ← cam_frame.tasks.get<view_task>() // 再从 camera 的内层 catalog 取具体 task
        record ← record_task{ /* 用 view 造 */ }
        // 如果这一帧没有 camera（entity 指针是 null），
        // compute_view 或者这里自己判断，返回空的 record —— 由值表达缺席


renderer_frame:
    tasks: catalog<begin_task, fence_task>

    renderer_frame(self: outer_catalog&):
        tasks ← catalog<begin_task, fence_task>{}


// -------- 每帧的驱动代码 --------

frame_catalog = catalog<camera_frame, foliage_frame, renderer_frame>

run_frame():
    frame ← frame_catalog{ /* 把 camera&, foliage&, renderer& 的指针
                              作为根信息传给需要的节点 */ }

    // 谁都不主动调，谁需要谁就递归拉起谁；
    // 或者显式 realize_all() 一遍把这一帧全部具体类型都摸一遍
    // （如果 catalog 决定保留这个便利操作）。

    foliage_result ← frame.get<foliage_frame>()
    if foliage_result.record.has_value():
        push(foliage_result.record->sender)

    // frame 出作用域，~catalog() 按真实构造顺序逆序销毁本帧全部任务，
    // camera/foliage/renderer 这三个长期对象本身毫发未损。
```
