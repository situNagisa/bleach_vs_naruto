模板参数改成 `Token` + `AvailableContainer`。池子只认这两套 concept：代币能怎么搬，可用容器能怎么进出。`_free` 改名为 `_available`。租约只 `get()` 出 `Token&`，不再假定它是指针。

---

```cpp
// lease_pool.h

// ------------------------------------------------------------ concepts

/// 在租约和可用容器之间搬来搬去的那一枚。
/// 空租约靠默认构造占位（_pool == nullptr 时不要读它）。
/// nothrow move：~lease / 转交才能维持 noexcept。
template <class T>
concept token =
    ::std::default_initializable<T> &&
    ::std::movable<T> &&
    ::std::is_nothrow_move_constructible_v<T> &&
    ::std::is_nothrow_move_assignable_v<T>;

/// 空闲代币的容器。接口按栈来：进 push_back，出 back + pop_back。
/// value_type 必须就是 Token。
///
/// 不要求 push_back nothrow（std::vector 不满足）。
/// 归还路径上若抛异常只能 terminate——和现在的 noexcept _put 一样。
/// 要避开分配：用定长栈，或 vector 先 reserve 到供给上限。
template <class C, class Token>
concept available_container =
    token<Token> &&
    requires(C& c, C const& cc, Token t) {
        typename C::value_type;
        requires ::std::same_as<typename C::value_type, Token>;
        c.push_back(::std::move(t));
        { c.back() } -> ::std::same_as<Token&>;
        c.pop_back();
        { cc.empty() } -> ::std::convertible_to<bool>;
        { cc.size() } -> ::std::convertible_to<::std::size_t>;
    };

// ------------------------------------------------------------ 池

template <class Token, class AvailableContainer = ::std::vector<Token>>
    requires token<Token> && available_container<AvailableContainer, Token>
struct lease_pool
{
    struct lease;
    struct waiter
    {
        waiter* _next = nullptr;
        void (*_complete)(waiter*, Token) noexcept = nullptr;
    };

    // ============================================================ 租约
    // 只可移动。_pool != nullptr ⇔ 握着一枚 Token。
    struct lease
    {
        lease_pool* _pool = nullptr;
        Token _token{};

        lease() = default;

        lease(lease_pool& pool, Token token) noexcept
            : _pool(&pool), _token(::std::move(token))
        {
        }

        lease(lease&& other) noexcept
            : _pool(::std::exchange(other._pool, nullptr))
            , _token(::std::move(other._token))
        {
        }

        auto operator=(lease&& other) noexcept -> lease&
        {
            auto discarded = ::std::move(*this);
            _pool = ::std::exchange(other._pool, nullptr);
            _token = ::std::move(other._token);
            return *this;
        }

        ~lease()
        {
            if (_pool != nullptr)
                _pool->_put(::std::move(_token));
        }

        [[nodiscard]] explicit operator bool() const noexcept { return _pool != nullptr; }
        [[nodiscard]] auto get() const noexcept -> Token const& { return _token; }
        [[nodiscard]] auto get() noexcept -> Token& { return _token; }
    };

    // ============================================================ acquire sender
    struct acquire_sender
    {
        using sender_concept = ::stdexec::sender_t;
        using completion_signatures = ::stdexec::completion_signatures<
            ::stdexec::set_value_t(lease),
            ::stdexec::set_stopped_t()>;

        lease_pool* _pool;

        template <class Receiver>
        struct operation : waiter
        {
            using operation_state_concept = ::stdexec::operation_state_t;

            struct on_stop
            {
                operation* _operation;
                auto operator()() const noexcept -> void
                {
                    if (!_operation->_pool->_unlink(_operation))
                        return; // 已被 _put 摘走，完成由那条路负责
                    _operation->_on_stop.reset();
                    ::stdexec::set_stopped(::std::move(_operation->_receiver));
                }
            };

            using token_type = ::stdexec::stop_token_of_t<::stdexec::env_of_t<Receiver>>;
            using callback_type = typename token_type::template callback_type<on_stop>;
            static constexpr bool _uses_stop_callback = !::stdexec::unstoppable_token<token_type>;

            struct no_callback {};
            using slot_type = ::std::conditional_t<
                _uses_stop_callback, ::std::optional<callback_type>, no_callback>;

            lease_pool* _pool;
            Receiver _receiver;
            [[no_unique_address]] slot_type _on_stop{};

            operation(lease_pool& pool, Receiver receiver)
                : _pool(&pool), _receiver(::std::move(receiver))
            {
                this->_complete = [](waiter* self, Token value) noexcept
                {
                    auto* const target = static_cast<operation*>(self);
                    if constexpr (_uses_stop_callback)
                        target->_on_stop.reset();
                    ::stdexec::set_value(
                        ::std::move(target->_receiver),
                        lease{*target->_pool, ::std::move(value)});
                };
            }

            operation(operation&&) = delete;
            auto operator=(operation&&) -> operation& = delete;

            auto start() & noexcept -> void
            {
                if (auto value = _pool->_take_or_enqueue(*this))
                {
                    ::stdexec::set_value(
                        ::std::move(_receiver), lease{*_pool, ::std::move(*value)});
                    return;
                }
                // 已入队。装 callback 时令牌可能已经停 → on_stop 立刻跑，之后不能碰 *this。
                if constexpr (_uses_stop_callback)
                {
                    _on_stop.emplace(
                        ::stdexec::get_stop_token(::stdexec::get_env(_receiver)),
                        on_stop{this});
                }
            }
        };

        template <::stdexec::receiver Receiver>
        [[nodiscard]] auto connect(Receiver receiver) const -> operation<Receiver>
        {
            return operation<Receiver>{*_pool, ::std::move(receiver)};
        }
    };

    // ============================================================ 池
    AvailableContainer _available{};
    ::std::mutex _mutex{};
    waiter* _head = nullptr;
    waiter* _tail = nullptr;
    ::std::size_t _supplied = 0;

    lease_pool() = default;

    explicit lease_pool(AvailableContainer available)
        : _available(::std::move(available)), _supplied(_available.size())
    {
    }

    lease_pool(lease_pool&&) = delete;
    auto operator=(lease_pool&&) -> lease_pool& = delete;

    ~lease_pool()
    {
        assert(_head == nullptr && "lease_pool: 还有等待者");
        assert(_available.size() == _supplied && "lease_pool: 还有租约没还");
    }

    /// 投入一枚代币。有人在等就直接转交。
    auto supply(Token value) -> void
    {
        {
            auto const guard = ::std::lock_guard{_mutex};
            ++_supplied;
        }
        _put(::std::move(value));
    }

    [[nodiscard]] auto acquire() noexcept -> acquire_sender { return acquire_sender{this}; }

    [[nodiscard]] auto try_acquire() -> ::std::optional<lease>
    {
        auto const guard = ::std::lock_guard{_mutex};
        if (_available.empty())
            return ::std::nullopt;
        auto value = ::std::move(_available.back());
        _available.pop_back();
        return ::std::optional<lease>{::std::in_place, *this, ::std::move(value)};
    }

    [[nodiscard]] auto available() -> ::std::size_t
    {
        auto const guard = ::std::lock_guard{_mutex};
        return _available.size();
    }

    [[nodiscard]] auto _take_or_enqueue(waiter& target) -> ::std::optional<Token>
    {
        auto const guard = ::std::lock_guard{_mutex};
        if (!_available.empty())
        {
            auto value = ::std::move(_available.back());
            _available.pop_back();
            return value;
        }
        target._next = nullptr;
        if (_tail == nullptr)
            _head = &target;
        else
            _tail->_next = &target;
        _tail = &target;
        return ::std::nullopt;
    }

    [[nodiscard]] auto _unlink(waiter* target) -> bool
    {
        auto const guard = ::std::lock_guard{_mutex};
        auto** link = &_head;
        auto* previous = static_cast<waiter*>(nullptr);
        while (*link != nullptr)
        {
            if (*link == target)
            {
                *link = target->_next;
                if (_tail == target)
                    _tail = previous;
                return true;
            }
            previous = *link;
            link = &(*link)->_next;
        }
        return false;
    }

    auto _put(Token value) noexcept -> void
    {
        auto* next = static_cast<waiter*>(nullptr);
        {
            auto const guard = ::std::lock_guard{_mutex};
            if (_head != nullptr)
            {
                next = _head;
                _head = next->_next;
                if (_head == nullptr)
                    _tail = nullptr;
            }
            else
            {
                _available.push_back(::std::move(value));
            }
        }
        if (next != nullptr)
            next->_complete(next, ::std::move(value));
    }
};
```

定长可用容器示例（满足 `available_container`，无堆分配）：

```cpp
template <class Token, ::std::size_t N>
    requires token<Token>
struct bounded_stack
{
    using value_type = Token;

    Token _data[N]{};
    ::std::size_t _size = 0;

    auto push_back(Token value) noexcept -> void
    {
        assert(_size < N);
        _data[_size++] = ::std::move(value);
    }
    auto back() noexcept -> Token& { return _data[_size - 1]; }
    auto pop_back() noexcept -> void { --_size; }
    [[nodiscard]] auto empty() const noexcept -> bool { return _size == 0; }
    [[nodiscard]] auto size() const noexcept -> ::std::size_t { return _size; }
};
```

**用法：**

```cpp
// Token = Slot*，对象在外面
::std::vector<Slot> items = /* 造 N 个，之后别扩容 */;
lease_pool<Slot*> pool;
for (auto& x : items) pool.supply(&x);

pool.acquire() | ::stdexec::then([](auto lease) { use(*lease.get()); });

// 换可用容器
lease_pool<Slot*, bounded_stack<Slot*, 8>> bounded;

// 无名额：Token 用 monostate，供给 N 次
lease_pool<::std::monostate> permits;
for (auto i = 0; i != 3; ++i) permits.supply({});
```

**concept 各自约束什么：**

| | 要求 | 不要求 |
|---|---|---|
| `token` | 默认构造、nothrow 移动 | 可解引用、可拷贝、是指针 |
| `available_container` | `value_type == Token`，栈式进出，`empty` / `size` | 随机访问、迭代器、nothrow `push_back` |

其余不变式与上一份相同：有 waiter 则 `_available` 必空；转交同一枚 `Token`；出锁再完成；池子不可移动。
