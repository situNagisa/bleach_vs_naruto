	built_entity

	entity
	{

		T _obj; // 实际的entity类型，即下面的xxx, yyy
		bool _is_built;
		task_pool;
		
		task_builder{
			entity&;
			
			emplace<T>(args...)
			{
				task_pool.emplace<T>(args);
			}
		}
		
		bool task_built() const { return _is_built; }
		auto build_task() 
		{
			assert(!_is_built);
			_obj.build_task(frame_context, task_builder)
			_is_built = true;
			return built_entity;
		}
		
		auto task<T>()
		{
			if(!_is_built)
				build_task();
			return task_pool.get<T>();
		}
	}
	
	range<entity> _entities; // 初始化一次
	
	auto entity<T>()
	{
		return _entities.get<T>(); // entity是一个概念，这里返回的是一个view
	}
